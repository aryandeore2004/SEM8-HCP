#include <iostream>
#include <omp.h>
#include <cstdlib>
#include <ctime>

using namespace std;

void autoInputArray(int arr[], int n) {
    for (int i = 0; i < n; i++) {
        arr[i] = rand() % 100;
    }
}

void serialBubbleSort(int arr[], int n) {
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                // swap
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

void parallelBubbleSort(int arr[], int n) {
    for (int i = 0; i < n; i++) {

        // Even phase
        #pragma omp parallel for
        for (int j = 0; j < n - 1; j += 2) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }

        // Odd phase
        #pragma omp parallel for
        for (int j = 1; j < n - 1; j += 2) {
            if (arr[j] > arr[j + 1]) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

void printArray(int arr[], int n) {
    for (int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }
    cout << endl;
}

int main() {
    int n;

    cout << "Enter number of elements: ";
    cin >> n;

    int arr[n], arr_serial[n], arr_parallel[n];

    // srand(static_cast<unsigned>(time(nullptr)));

    autoInputArray(arr, n);

    // cout << "Generated " << n << " random numbers:\n";
    // printArray(arr, n);

    for (int i = 0; i < n; i++) {
        arr_serial[i] = arr[i];     // copy for serial
        arr_parallel[i] = arr[i];   // copy for parallel
    }

    double start_serial = omp_get_wtime();

    serialBubbleSort(arr_serial, n);

    double end_serial = omp_get_wtime();

    double start_parallel = omp_get_wtime();

    parallelBubbleSort(arr_parallel, n);

    double end_parallel = omp_get_wtime();
    
    cout << "\n===== SERIAL SORT =====\n";
    // printArray(arr_serial, n);
    cout << "Time: " << (end_serial - start_serial) << " sec\n";

    cout << "\n===== PARALLEL SORT =====\n";
    // printArray(arr_parallel, n);
    cout << "Time: " << (end_parallel - start_parallel) << " sec\n";
    
    return 0;
}
